    Imports:
      import json
      import logging
      from datetime import datetime, timezone
      from typing import Optional, TYPE_CHECKING

      if TYPE_CHECKING:
          pass  # no circular imports here

      from src.inbox.models import InboxItem, ActionItem, CommType, SourceType
      # Lazy import of call_haiku to avoid loading settings at module import time
      # Use: from src.shared.llm.claude import call_haiku inside the function

    CLASSIFICATION_SCHEMA = {
        "type": "object",
        "required": ["comm_type", "urgency", "summary", "action_items"],
        "properties": {
            "comm_type": {
                "type": "string",
                "enum": ["decision", "action-item", "FYI", "question", "escalation"]
            },
            "urgency": {"type": "integer", "minimum": 1, "maximum": 5},
            "summary": {"type": "string"},
            "project_id": {"type": ["string", "null"]},
            "project_name": {"type": ["string", "null"]},
            "action_items": {
                "type": "array",
                "items": {
                    "type": "object",
                    "required": ["assignee", "description", "priority"],
                    "properties": {
                        "assignee": {"type": "string"},
                        "deadline": {"type": ["string", "null"]},
                        "priority": {"type": "string", "enum": ["low", "medium", "high"]},
                        "description": {"type": "string"}
                    }
                }
            }
        }
    }

    SYSTEM_PROMPT = """You are a construction project communication classifier for GigAI.
    Analyze the given construction project communication and:
    1. Classify comm_type: decision (a choice was made), action-item (something must be done),
       FYI (informational only), question (requires an answer), escalation (urgent issue requiring senior attention)
    2. Assign urgency 1-5: 1=low/routine, 3=moderate attention needed, 5=critical/stop-work
    3. Write a 1-2 sentence summary
    4. Extract action items: for each task/todo/followup mentioned, extract assignee, deadline (ISO8601 or null), priority, description
    5. Route to a project: from the project list provided, select the best matching project_id and project_name based on content; use null if no match
    Return valid JSON matching the schema. Be precise."""

    # Build compact project list string from src/data/projects module
    # Called once at first classify_communication() call, cached in module variable
    _PROJECT_CONTEXT: str | None = None

    def _get_project_context() -> str:
        global _PROJECT_CONTEXT
        if _PROJECT_CONTEXT is None:
            from src.data.projects import PROJECTS  # lazy import
            lines = [f"{p.project_id}: {p.name}" for p in PROJECTS]
            _PROJECT_CONTEXT = "Available projects:\n" + "\n".join(lines)
        return _PROJECT_CONTEXT

    def classify_communication(
        raw_text: str,
        source: str,  # SourceType
        sender: str,
        sender_email: Optional[str] = None,
        subject: Optional[str] = None,
        source_url: Optional[str] = None,
        received_at: Optional[datetime] = None,
    ) -> InboxItem:
        from src.shared.llm.claude import call_haiku

        if received_at is None:
            received_at = datetime.now(timezone.utc)

        project_context = _get_project_context()
        prompt = f"""Communication to classify:
    Subject: {subject or 'N/A'}
    From: {sender} <{sender_email or 'unknown'}>
    Source: {source}
    Body:
    {raw_text[:3000]}

    {project_context}"""

        try:
            raw_json = call_haiku(
                prompt=prompt,
                system=SYSTEM_PROMPT,
                output_schema=CLASSIFICATION_SCHEMA,
                max_tokens=1024,
                run_type="inbox_classification",
            )
            data = json.loads(raw_json)

            # Parse action items
            action_items = []
            for ai_data in data.get("action_items", []):
                deadline = None
                if ai_data.get("deadline"):
                    try:
                        deadline = datetime.fromisoformat(ai_data["deadline"].replace("Z", "+00:00"))
                    except (ValueError, AttributeError):
                        deadline = None
                action_items.append(ActionItem(
                    assignee=ai_data.get("assignee", "Unassigned"),
                    deadline=deadline,
                    priority=ai_data.get("priority", "medium"),
                    description=ai_data.get("description", ""),
                ))

            return InboxItem(
                source=source,
                comm_type=data["comm_type"],
                urgency=max(1, min(5, int(data.get("urgency", 3)))),
                summary=data.get("summary", raw_text[:200]),
                raw_text=raw_text,
                project_id=data.get("project_id"),
                project_name=data.get("project_name"),
                sender=sender,
                sender_email=sender_email,
                subject=subject,
                source_url=source_url,
                action_items=action_items,
                received_at=received_at,
                classified_at=datetime.now(timezone.utc),
            )

        except Exception as exc:
            logger.warning("Classification failed for comm from %s: %s — using fallback", sender, exc)
            return InboxItem(
                source=source,
                comm_type="FYI",
                urgency=2,
                summary=raw_text[:200] if raw_text.strip() else "Communication received",
                raw_text=raw_text,
                sender=sender,
                sender_email=sender_email,
                subject=subject,
                source_url=source_url,
                action_items=[],
                received_at=received_at,
                classified_at=None,
            )

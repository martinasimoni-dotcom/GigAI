```python
"""
Proposal generator — DI-02.
Calls Sonnet 4 with enriched context and signals, validates with Pydantic,
retries on ValidationError or JSONDecodeError.
No module-level config.settings import.
"""
from __future__ import annotations
import json
import logging
from pathlib import Path
from typing import TYPE_CHECKING

from tenacity import retry, stop_after_attempt, wait_exponential
from pydantic import ValidationError

from src.shared.models.proposals import Action, Proposal

if TYPE_CHECKING:
    from src.system.domain_processing.processor import ProcessingResult

logger = logging.getLogger(__name__)

# Load prompt file relative to this module — avoids config.settings dependency
_PROMPT_PATH = Path(__file__).parent.parent.parent.parent / "config" / "prompts" / "proposal.txt"


def _build_user_prompt(processing_result: "ProcessingResult") -> str:
    """Serialize ProcessingResult into the user message for Sonnet."""
    # Build a readable context dict from processing_result
    event = processing_result.event.event
    context = {
        "event_type": event.event_type,
        "summary": event.summary,
        "material_original": event.material_original,
        "material_new": event.material_new,
        "location": event.location,
        "quantity": event.quantity,
        "estimated_cost": event.estimated_cost,
        "signals": [s.signal_type for s in processing_result.signals],
        "triggered_rules": processing_result.policy_result.triggered_rules,
        "time_conflict": processing_result.time_result.has_conflict,
        "knowledge_chunks": [c.get("content", "") for c in (processing_result.event.knowledge_chunks or [])],
        "historical_matches": [
            {"content": m.content, "outcome": m.outcome, "similarity": m.similarity}
            for m in (processing_result.event.historical_matches or [])
        ],
        "floor_plan_data": processing_result.event.floor_plan_data,
    }
    return json.dumps(context, indent=2, default=str)


@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=2, max=10),
    reraise=True,
)
def generate_proposal(processing_result: "ProcessingResult") -> Proposal:
    """
    Generate a coordination proposal using Sonnet 4.

    Calls call_sonnet with the proposal.txt system prompt and serialized
    ProcessingResult context. Parses and validates the JSON response.
    Retries up to 3 times on ValidationError or json.JSONDecodeError.

    Args:
        processing_result: Full ProcessingResult from domain processing pipeline.

    Returns:
        Validated Proposal object with Actions, confidence_score, and recommendation.

    Raises:
        tenacity.RetryError: After 3 failed attempts.
    """
    # Import here (not at module level) to avoid config.settings singleton trigger
    from src.shared.llm.claude import call_sonnet
    from src.system.decision_intelligence.confidence_scorer import score_proposal

    system_prompt = _PROMPT_PATH.read_text(encoding="utf-8")
    user_prompt = _build_user_prompt(processing_result)

    raw = call_sonnet(prompt=user_prompt, system=system_prompt)

    # Parse and validate — tenacity retries if these raise
    data = json.loads(raw)

    # Map recommended_actions → Action objects
    actions = []
    for raw_action in data.get("recommended_actions", []):
        action_type = raw_action.get("action_type", "task")
        if action_type not in ("email", "task", "calendar", "drawing"):
            action_type = "task"
        actions.append(Action(
            action_type=action_type,
            action_data=raw_action,
        ))

    # Score confidence
    historical = processing_result.event.historical_matches or []
    confidence_result = score_proposal(processing_result, historical)

    proposal = Proposal(
        event_id=processing_result.event.event.event_id,
        alert={
            "summary": data.get("event_summary", ""),
            "stakeholders": data.get("affected_stakeholders", []),
            "confidence_rationale": data.get("confidence_rationale", ""),
        },
        actions=actions,
        confidence_score=confidence_result.score / 100.0,  # stored as 0-1
        recommendation=confidence_result.recommendation,
    )

    logger.info({
        "event": "proposal_generated",
        "event_id": proposal.event_id,
        "actions": len(proposal.actions),
        "confidence_score": proposal.confidence_score,
        "recommendation": proposal.recommendation,
    })

    return proposal
```

NOTE: The retry decorator wraps the entire function including JSON parse and Pydantic validation — json.JSONDecodeError and ValidationError will both trigger retries since the function raises before returning. However, call_sonnet already has its own @retry inside claude.py. The outer @retry on generate_proposal catches parse/validation failures that call_sonnet success does not cover.

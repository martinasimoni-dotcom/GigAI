**src/output/notifications/dashboard_notifier.py:**

Use pure in-memory asyncio.Queue for SSE. No external dependencies beyond asyncio and json.

```python
from __future__ import annotations
import asyncio
import json
import logging
from typing import AsyncGenerator

logger = logging.getLogger(__name__)

# Module-level event queue — shared across all connected SSE clients.
# notify_dashboard() pushes events here; get_event_stream() consumes them.
_event_queue: asyncio.Queue = asyncio.Queue()


def notify_dashboard(proposal_response: dict) -> None:
    """Push a proposal response dict to the SSE event queue (non-blocking)."""
    try:
        _event_queue.put_nowait(proposal_response)
        logger.info("Dashboard notifier: queued proposal %s", proposal_response.get("id"))
    except asyncio.QueueFull:
        logger.warning("Dashboard SSE queue full — dropping event")


async def get_event_stream() -> AsyncGenerator[str, None]:
    """
    Async generator for Server-Sent Events.
    Yields SSE-formatted strings: 'data: {json}\\n\\n'
    Used by FastAPI GET /api/events endpoint with EventSourceResponse.
    """
    while True:
        try:
            data = await asyncio.wait_for(_event_queue.get(), timeout=15.0)
            payload = json.dumps(data)
            yield f"data: {payload}\n\n"
            _event_queue.task_done()
        except asyncio.TimeoutError:
            # Keep-alive heartbeat
            yield ": heartbeat\n\n"
        except Exception as e:
            logger.error("SSE stream error: %s", e)
            break
```

```python
from __future__ import annotations
import logging
import os
import urllib.request
import json
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from src.shared.models.proposals import Proposal

logger = logging.getLogger(__name__)


def notify_acc(proposal: "Proposal") -> None:
    """Post an ACC issue/comment as a secondary notification for the proposal."""
    token = os.getenv("ACC_TOKEN")
    account_id = os.getenv("ACC_ACCOUNT_ID")
    if not token or not account_id:
        logger.warning("ACC_TOKEN or ACC_ACCOUNT_ID not set — skipping ACC notification for proposal %s", proposal.proposal_id)
        return
    try:
        payload = json.dumps({
            "title": f"GigAI Proposal: {proposal.alert.get('title', 'Change')}",
            "description": f"Confidence: {round(proposal.confidence_score * 100, 1)}% | Recommendation: {proposal.recommendation}",
        }).encode()
        url = f"https://developer.api.autodesk.com/construction/issues/v1/containers/{account_id}/quality-issues"
        req = urllib.request.Request(url, data=payload, headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"}, method="POST")
        with urllib.request.urlopen(req, timeout=10):
            pass
        logger.info("ACC notification sent for proposal %s", proposal.proposal_id)
    except Exception as e:
        logger.error("ACC notifier failed for proposal %s: %s", proposal.proposal_id, e)
```

```python
from __future__ import annotations
import logging
import os
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from src.shared.models.proposals import Proposal

logger = logging.getLogger(__name__)

_DDL = """
CREATE TABLE IF NOT EXISTS decisions (
    id SERIAL PRIMARY KEY,
    proposal_id TEXT NOT NULL,
    event_id TEXT,
    decision TEXT NOT NULL,
    reason TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
)
"""


def record_decision(
    proposal_id: str,
    decision: str,
    reason: str | None,
    proposal: "Proposal",
) -> None:
    """
    Log a PM decision (accept/reject) to PostgreSQL and embed as a historical pattern.

    - Inserts into decisions table (CREATE TABLE IF NOT EXISTS on first call).
    - Embeds the decision text into pgvector knowledge_chunks with source="decision_patterns".
    - If DATABASE_URL is not set or DB call fails, logs and returns without raising.
    - If VOYAGE_API_KEY is not set, embedding step is skipped with a warning.
    """
    db_url = os.getenv("DATABASE_URL")
    if not db_url:
        logger.warning("DATABASE_URL not set — skipping DB record for proposal %s", proposal_id)
    else:
        try:
            from src.shared.db.postgres import get_connection, release_connection
            conn = get_connection()
            try:
                with conn.cursor() as cur:
                    cur.execute(_DDL)
                    cur.execute(
                        "INSERT INTO decisions (proposal_id, event_id, decision, reason) VALUES (%s, %s, %s, %s)",
                        (proposal_id, proposal.event_id, decision, reason),
                    )
                conn.commit()
                logger.info("Decision recorded: proposal=%s decision=%s", proposal_id, decision)
            finally:
                release_connection(conn)
        except Exception as exc:
            logger.error("Failed to record decision for proposal %s: %s", proposal_id, exc)

    # Embed decision pattern for future historical retrieval
    voyage_key = os.getenv("VOYAGE_API_KEY")
    if not voyage_key:
        logger.warning("VOYAGE_API_KEY not set — skipping pgvector embedding for proposal %s", proposal_id)
        return
    try:
        from src.shared.db.vector_store import embed_and_store
        title = proposal.alert.get("title", "material change")
        reason_text = reason or "no reason given"
        decision_text = f"{decision}: {title} — {reason_text}"
        embed_and_store(
            texts=[decision_text],
            source="decision_patterns",
            metadata={
                "proposal_id": proposal_id,
                "event_id": proposal.event_id,
                "decision": decision,
            },
        )
        logger.info("Decision embedded in pgvector: proposal=%s", proposal_id)
    except Exception as exc:
        logger.error("Failed to embed decision for proposal %s: %s", proposal_id, exc)
```

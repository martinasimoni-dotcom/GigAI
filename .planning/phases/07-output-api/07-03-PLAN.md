from __future__ import annotations
import asyncio
import logging
from typing import TYPE_CHECKING

from pydantic import BaseModel

from src.output.action_gateway.action_result import ActionResult

if TYPE_CHECKING:
    from src.shared.models.proposals import Action, Proposal

logger = logging.getLogger(__name__)


class ExecutionResult(BaseModel):
    """Result of executing all actions for a proposal."""
    proposal_id: str
    results: list[ActionResult]
    success_count: int
    failure_count: int


def _get_executor(action_type: str):
    """Return the correct executor function for an action_type."""
    if action_type == "email":
        from src.output.action_gateway.gmail_executor import execute_gmail_action
        return execute_gmail_action
    elif action_type == "task":
        from src.output.action_gateway.acc_executor import execute_acc_action
        return execute_acc_action
    elif action_type == "calendar":
        from src.output.action_gateway.calendar_executor import execute_calendar_action
        return execute_calendar_action
    elif action_type == "drawing":
        from src.output.action_gateway.document_executor import execute_document_action
        return execute_document_action
    return None


async def _run_action(action: "Action") -> ActionResult:
    """Run a single action executor, wrapping exceptions as failed ActionResult."""
    executor = _get_executor(action.action_type)
    if executor is None:
        return ActionResult(
            action_type=action.action_type,
            status="skipped",
            message=f"unknown action_type: {action.action_type}",
        )
    try:
        # Executors are sync — run in thread pool to avoid blocking the event loop
        result = await asyncio.to_thread(executor, action)
        return result
    except Exception as exc:
        logger.error("Executor raised for action %s (%s): %s", action.action_id, action.action_type, exc)
        return ActionResult(
            action_type=action.action_type,
            status="failed",
            message="Executor raised exception",
            error=str(exc),
        )


async def execute_actions(proposal: "Proposal") -> ExecutionResult:
    """
    Execute all actions in a proposal in parallel.

    Uses asyncio.gather with return_exceptions=False (exceptions caught in _run_action).
    Partial failures are captured — other executors still complete.
    """
    tasks = [_run_action(action) for action in proposal.actions]
    results: list[ActionResult] = await asyncio.gather(*tasks)

    success_count = sum(1 for r in results if r.status == "success")
    failure_count = sum(1 for r in results if r.status == "failed")

    logger.info(
        "execute_actions: proposal=%s success=%d failed=%d skipped=%d",
        proposal.proposal_id,
        success_count,
        failure_count,
        len(results) - success_count - failure_count,
    )
    return ExecutionResult(
        proposal_id=proposal.proposal_id,
        results=list(results),
        success_count=success_count,
        failure_count=failure_count,
    )
```

from __future__ import annotations
import asyncio
import logging
from typing import Optional

from fastapi import APIRouter, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

logger = logging.getLogger(__name__)

router = APIRouter(prefix="/api")

# In-memory proposal store — populated by store_proposal()
_proposals: list[dict] = []


class DecisionRequest(BaseModel):
    decision: str   # "accept" or "reject"
    reason: Optional[str] = None


def store_proposal(proposal_response: dict) -> None:
    """
    Store a built proposal response and push SSE notification.
    Called by the pipeline after proposal generation.
    """
    _proposals.append(proposal_response)
    try:
        from src.output.notifications.dashboard_notifier import notify_dashboard
        notify_dashboard(proposal_response)
    except Exception as exc:
        logger.error("Failed to notify dashboard: %s", exc)
    logger.info("Proposal stored: id=%s", proposal_response.get("id"))


@router.get("/health")
async def health() -> dict:
    return {"status": "ok", "version": "0.1.0"}


@router.get("/proposals")
async def get_proposals() -> list[dict]:
    return _proposals


@router.post("/proposals/{proposal_id}/decision")
async def post_decision(proposal_id: str, body: DecisionRequest) -> dict:
    # Find the proposal in memory
    proposal_data = next((p for p in _proposals if p.get("id") == proposal_id), None)
    if proposal_data is None:
        raise HTTPException(status_code=404, detail=f"Proposal {proposal_id} not found")

    # Reconstruct the Proposal model for gateway/feedback
    try:
        from src.shared.models.proposals import Proposal
        proposal = Proposal(
            proposal_id=proposal_data["id"],
            event_id=proposal_data["event_id"],
            alert=proposal_data["alert"],
            actions=[],           # actions already formatted — gateway re-executes from original
            confidence_score=proposal_data["confidence_score"] / 100.0,
            recommendation=proposal_data["recommendation"],
        )
    except Exception as exc:
        logger.error("Failed to reconstruct proposal %s: %s", proposal_id, exc)
        raise HTTPException(status_code=500, detail=f"Proposal reconstruction failed: {exc}")

    # Always record the decision
    try:
        from src.output.feedback.feedback_loop import record_decision
        record_decision(proposal_id, body.decision, body.reason, proposal)
    except Exception as exc:
        logger.warning("record_decision failed (non-fatal): %s", exc)

    if body.decision == "accept":
        # Re-build proposal with original actions from in-memory data for execution
        try:
            from src.shared.models.proposals import Proposal, Action
            actions = [Action(**a) for a in proposal_data.get("actions", [])]
            full_proposal = proposal.model_copy(update={"actions": actions})
            from src.output.action_gateway.gateway import execute_actions
            execution_result = await execute_actions(full_proposal)
            return execution_result.model_dump()
        except Exception as exc:
            logger.error("execute_actions failed for proposal %s: %s", proposal_id, exc)
            raise HTTPException(status_code=500, detail=f"Execution failed: {exc}")

    elif body.decision == "reject":
        return {"status": "rejected", "proposal_id": proposal_id}

    else:
        raise HTTPException(status_code=422, detail=f"Unknown decision: {body.decision!r}. Must be 'accept' or 'reject'")


@router.get("/events")
async def get_events():
    """Server-Sent Events stream for real-time proposal notifications."""
    from src.output.notifications.dashboard_notifier import get_event_stream
    return StreamingResponse(get_event_stream(), media_type="text/event-stream")
```

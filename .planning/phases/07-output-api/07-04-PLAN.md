**src/api/middleware.py:**

Implement Starlette BaseHTTPMiddleware for API key auth. CORS is added via
`app.add_middleware(CORSMiddleware, ...)` in main.py (not here). Exception handler
is an app-level decorator also in main.py.

```python
from __future__ import annotations
import json
import logging
import os

from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import Response

logger = logging.getLogger(__name__)

# Paths that bypass API key auth — always public
_PUBLIC_PATHS = {"/health", "/api/health"}


class APIKeyMiddleware(BaseHTTPMiddleware):
    """
    Simple API key authentication middleware.

    Checks X-API-Key header against API_KEY environment variable.
    - If API_KEY is not set: auth is disabled entirely (dev mode).
    - Paths in _PUBLIC_PATHS bypass auth even when API_KEY is set.
    - Non-matching key returns HTTP 401 JSON response.
    """

    async def dispatch(self, request: Request, call_next) -> Response:
        api_key = os.getenv("API_KEY")
        if not api_key:
            # Dev mode — no key configured, all requests pass through
            return await call_next(request)

        path = request.url.path
        if path in _PUBLIC_PATHS:
            return await call_next(request)

        provided_key = request.headers.get("X-API-Key", "")
        if provided_key != api_key:
            logger.warning("Unauthorized request to %s — invalid or missing X-API-Key", path)
            body = json.dumps({"error": "Unauthorized", "type": "AuthenticationError"})
            return Response(content=body, status_code=401, media_type="application/json")

        return await call_next(request)
```

```python
"""
GigAI FastAPI application entry point.
Registers all input-layer webhook routers, output API routes, middleware, and exposes the ASGI app.
"""
import logging

from fastapi import FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse

from src.input.webhooks import fireflies, acc
from src.api.routes import router as api_router
from src.api.middleware import APIKeyMiddleware

logger = logging.getLogger(__name__)

app = FastAPI(
    title="GigAI",
    description="Material Change Coordination Automation",
    version="1.0.0",
)

# CORS — allow all origins in development; restrict in production via API_KEY
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# API key auth — no-op if API_KEY env var not set (dev mode)
app.add_middleware(APIKeyMiddleware)


@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception) -> JSONResponse:
    logger.error("Unhandled exception on %s %s: %s", request.method, request.url.path, exc)
    return JSONResponse(
        status_code=500,
        content={"error": str(exc), "type": type(exc).__name__},
    )


# Input-layer webhook routers (Phase 1)
app.include_router(fireflies.router)
app.include_router(acc.router)

# Output API router (Phase 7)
app.include_router(api_router)


@app.get("/health")
async def health() -> dict:
    """Health check endpoint (legacy — Phase 1). GET /api/health also available."""
    return {"status": "ok"}
```

**src/output/action_gateway/acc_executor.py:**
```python
from __future__ import annotations
import logging, os
from typing import TYPE_CHECKING
from src.output.action_gateway.action_result import ActionResult
if TYPE_CHECKING:
    from src.shared.models.proposals import Action

logger = logging.getLogger(__name__)

def execute_acc_action(action: "Action") -> ActionResult:
    token = os.getenv("ACC_TOKEN")
    account_id = os.getenv("ACC_ACCOUNT_ID")
    if not token or not account_id:
        logger.warning("ACC_TOKEN or ACC_ACCOUNT_ID not set — returning stub success")
        return ActionResult(
            action_type=action.action_type,
            status="success",
            message=f"stub: would create ACC {action.action_type} — {action.action_data.get('title', 'untitled')}",
        )
    try:
        import urllib.request, json as _json
        url = f"https://developer.api.autodesk.com/construction/issues/v1/containers/{account_id}/quality-issues"
        payload = _json.dumps({"title": action.action_data.get("title", "GigAI Task"), "description": action.action_data.get("description", "")}).encode()
        req = urllib.request.Request(url, data=payload, headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"}, method="POST")
        with urllib.request.urlopen(req, timeout=10) as resp:
            result = _json.loads(resp.read())
        return ActionResult(action_type=action.action_type, status="success", message=f"ACC task created: {result.get('id', 'unknown')}")
    except Exception as e:
        logger.error("ACC executor failed: %s", e)
        return ActionResult(action_type=action.action_type, status="failed", message="ACC API error", error=str(e))
```

**src/output/action_gateway/gmail_executor.py:**
```python
from __future__ import annotations
import logging, os, base64
from typing import TYPE_CHECKING
from src.output.action_gateway.action_result import ActionResult
if TYPE_CHECKING:
    from src.shared.models.proposals import Action

logger = logging.getLogger(__name__)

def execute_gmail_action(action: "Action") -> ActionResult:
    creds_json = os.getenv("GMAIL_CREDENTIALS_JSON")
    if not creds_json:
        logger.warning("GMAIL_CREDENTIALS_JSON not set — returning stub success")
        to = action.action_data.get("to", "unknown")
        subject = action.action_data.get("subject", "GigAI Notification")
        return ActionResult(action_type=action.action_type, status="success", message=f"stub: would send email to {to} subject '{subject}'")
    try:
        import json as _json
        from google.oauth2.service_account import Credentials
        from googleapiclient.discovery import build
        import email.mime.text as _mime
        creds_data = _json.loads(creds_json)
        creds = Credentials.from_service_account_info(creds_data, scopes=["https://www.googleapis.com/auth/gmail.send"])
        service = build("gmail", "v1", credentials=creds)
        to = action.action_data.get("to", "")
        subject = action.action_data.get("subject", "GigAI Notification")
        body = action.action_data.get("body", "")
        msg = _mime.MIMEText(body)
        msg["to"] = to
        msg["subject"] = subject
        raw = base64.urlsafe_b64encode(msg.as_bytes()).decode()
        service.users().messages().send(userId="me", body={"raw": raw}).execute()
        return ActionResult(action_type=action.action_type, status="success", message=f"Email sent to {to}")
    except Exception as e:
        logger.error("Gmail executor failed: %s", e)
        return ActionResult(action_type=action.action_type, status="failed", message="Gmail API error", error=str(e))
```

```python
from __future__ import annotations
import logging, os
from typing import TYPE_CHECKING
from src.output.action_gateway.action_result import ActionResult
if TYPE_CHECKING:
    from src.shared.models.proposals import Action

logger = logging.getLogger(__name__)

def execute_calendar_action(action: "Action") -> ActionResult:
    creds_json = os.getenv("GMAIL_CREDENTIALS_JSON")
    cal_id = os.getenv("GOOGLE_CALENDAR_ID")
    if not creds_json or not cal_id:
        logger.warning("Calendar credentials not set — returning stub success")
        summary = action.action_data.get("summary", "GigAI Follow-up")
        return ActionResult(action_type=action.action_type, status="success", message=f"stub: would create calendar event '{summary}'")
    try:
        import json as _json
        from datetime import datetime, timezone, timedelta
        from google.oauth2.service_account import Credentials
        from googleapiclient.discovery import build
        creds_data = _json.loads(creds_json)
        creds = Credentials.from_service_account_info(creds_data, scopes=["https://www.googleapis.com/auth/calendar"])
        service = build("calendar", "v3", credentials=creds)
        start_str = action.action_data.get("start")
        if start_str:
            start = datetime.fromisoformat(start_str)
        else:
            start = datetime.now(timezone.utc) + timedelta(days=3)
        end = start + timedelta(days=1)
        event = {
            "summary": action.action_data.get("summary", "GigAI Follow-up"),
            "description": action.action_data.get("description", ""),
            "start": {"dateTime": start.isoformat(), "timeZone": "UTC"},
            "end": {"dateTime": end.isoformat(), "timeZone": "UTC"},
        }
        result = service.events().insert(calendarId=cal_id, body=event, singleEvents=True).execute()
        return ActionResult(action_type=action.action_type, status="success", message=f"Calendar event created: {result.get('id')}")
    except Exception as e:
        logger.error("Calendar executor failed: %s", e)
        return ActionResult(action_type=action.action_type, status="failed", message="Calendar API error", error=str(e))
```

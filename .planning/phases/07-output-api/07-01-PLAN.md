    ```python
    def build_proposal_response(proposal: Proposal) -> dict:
        return {
            "id": proposal.proposal_id,
            "event_id": proposal.event_id,
            "alert": proposal.alert,
            "actions": [
                {
                    "action_id": a.action_id,
                    "action_type": a.action_type,
                    "action_data": a.action_data,
                    "status": a.status,
                }
                for a in proposal.actions
            ],
            "confidence_score": round(proposal.confidence_score * 100),
            "recommendation": proposal.recommendation,
            "created_at": proposal.created_at.isoformat(),
        }
    ```

    ```python
    """ACC Executor (OUT-03): creates issues via ACC Issues API."""
    import os
    from src.output.action_gateway import ActionResult
    from src.shared.clients import acc as acc_client
    from src.shared.models.proposals import Action

    async def execute_acc_action(action: Action) -> ActionResult:
        project_id = os.getenv("ACC_PROJECT_ID")
        container_id = os.getenv("ACC_ISSUES_CONTAINER_ID")
        if not project_id or not container_id:
            raise RuntimeError(
                "ACC_PROJECT_ID and ACC_ISSUES_CONTAINER_ID must be set in .env"
            )
        data = action.action_data
        issue = acc_client.create_issue(
            project_id=project_id,
            container_id=container_id,
            title=data["title"],
            description=data["description"],
            assignee_id=data.get("assignee_id"),
            due_date=data.get("due_date"),
        )
        issue_id = issue.get("id", "unknown")
        return ActionResult(
            action_type="task",
            status="success",
            message=f"ACC issue created: {issue_id}",
        )
    ```

    ```python
    """Gmail Executor (OUT-04): sends emails via Gmail API v1."""
    import base64
    import email.mime.text
    import os
    from email.mime.multipart import MIMEMultipart
    from email.mime.text import MIMEText

    import google.auth.transport.requests
    import google.oauth2.credentials
    import googleapiclient.discovery

    from src.output.action_gateway import ActionResult
    from src.shared.models.proposals import Action


    def _build_gmail_service():
        client_id = os.getenv("GMAIL_CLIENT_ID")
        client_secret = os.getenv("GMAIL_CLIENT_SECRET")
        refresh_token = os.getenv("GMAIL_REFRESH_TOKEN")
        if not client_id or not client_secret or not refresh_token:
            raise RuntimeError(
                "GMAIL_CLIENT_ID, GMAIL_CLIENT_SECRET, and GMAIL_REFRESH_TOKEN must be set in .env"
            )
        creds = google.oauth2.credentials.Credentials(
            token=None,
            refresh_token=refresh_token,
            token_uri="https://oauth2.googleapis.com/token",
            client_id=client_id,
            client_secret=client_secret,
        )
        creds.refresh(google.auth.transport.requests.Request())
        return googleapiclient.discovery.build("gmail", "v1", credentials=creds)


    async def execute_gmail_action(action: Action) -> ActionResult:
        service = _build_gmail_service()
        data = action.action_data
        to = data["to"]
        msg = MIMEText(data["body"], "plain")
        msg["To"] = to
        msg["Subject"] = data["subject"]
        raw = base64.urlsafe_b64encode(msg.as_bytes()).decode()
        result = service.users().messages().send(userId="me", body={"raw": raw}).execute()
        msg_id = result.get("id", "unknown")
        return ActionResult(
            action_type="email",
            status="success",
            message=f"Email sent to {to} (message_id={msg_id})",
        )
    ```

    ```python
    import google.auth.transport.requests
    import google.oauth2.credentials
    import googleapiclient.discovery

    from src.output.action_gateway import ActionResult
    from src.shared.models.proposals import Action


    def _build_calendar_service():
        client_id = os.getenv("GMAIL_CLIENT_ID")
        client_secret = os.getenv("GMAIL_CLIENT_SECRET")
        refresh_token = os.getenv("GMAIL_REFRESH_TOKEN")
        if not client_id or not client_secret or not refresh_token:
            raise RuntimeError(
                "GMAIL_CLIENT_ID, GMAIL_CLIENT_SECRET, and GMAIL_REFRESH_TOKEN must be set in .env"

        creds = google.oauth2.credentials.Credentials(
            token=None,
            refresh_token=refresh_token,
            token_uri="https://oauth2.googleapis.com/token",
            client_id=client_id,
            client_secret=client_secret,
        )
        creds.refresh(google.auth.transport.requests.Request())
        return googleapiclient.discovery.build("calendar", "v3", credentials=creds)


    async def execute_calendar_action(action: Action) -> ActionResult:

        event_id = result.get("id", "unknown")
    ```

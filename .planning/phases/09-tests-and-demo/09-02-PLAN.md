**Execution loop and scoring:**
```python
def run_validation():
    passed = 0
    failed = []

    for item in QUERIES:
        results = search(item["query"], top_k=1)
        if results and results[0].get("source", "").startswith(item["expected_source"]):
            passed += 1
            status = "PASS"
        else:
            failed.append(item["label"])
            status = "FAIL"
            actual = results[0].get("source", "no results") if results else "no results"
            print(f"  FAIL [{item['label']}] expected={item['expected_source']} got={actual}")

    total = len(QUERIES)
    recall = passed / total
    print(f"\nRecall: {passed}/{total} = {recall:.1%}")

    if failed:
        print(f"Failed queries: {', '.join(failed)}")

    if recall < 0.90:
        print(f"BELOW THRESHOLD: {recall:.1%} < 90%")
        sys.exit(1)
    else:
        print("PASS: recall >= 90%")
        sys.exit(0)
```
